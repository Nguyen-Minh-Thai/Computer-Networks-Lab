\subsubsection*{Exercise 1:}
\noindent\textit{Create a program that connects to a web server and downloads the homepage of this website to local computer.\\
Ex.: You can test your program with: www.google.com}

\begin{solutionbox}[title=Solution for Exercise 1: Download Homepage]
\textbf{1. Implementation Overview (\texttt{DownloadHomepage.java}):}
\begin{itemize}
    \item \textbf{TCP Connection:} Creates a client socket connecting to port 80 (HTTP) of \texttt{www.google.com}.
    \item \textbf{HTTP Request:} Sends a standard HTTP/1.0 GET request:
    \begin{verbatim}
    GET / HTTP/1.0\r\n
    Host: www.google.com\r\n
    User-Agent: Mozilla/5.0\r\n\r\n
    \end{verbatim}
    Using HTTP/1.0 ensures the server automatically closes the connection once transmission is finished, so the stream reaches EOF (\texttt{-1}).
    \item \textbf{Header \& Body Parsing:} Reads all incoming bytes into a buffer, detects the HTTP separator sequence \texttt{\textbackslash r\textbackslash n\textbackslash r\textbackslash n}, prints the HTTP response headers to console, and writes the remaining body bytes into \texttt{homepage.html}.
\end{itemize}

\textbf{2. Key Source Code:}
\begin{lstlisting}
try (Socket socket = new Socket(host, 80)) {
    OutputStream out = socket.getOutputStream();
    String request = "GET / HTTP/1.0\r\nHost: " + host + "\r\nUser-Agent: Mozilla/5.0\r\n\r\n";
    out.write(request.getBytes("ISO-8859-1"));
    out.flush();

    InputStream in = socket.getInputStream();
    ByteArrayOutputStream buf = new ByteArrayOutputStream();
    byte[] chunk = new byte[4096];
    int n;
    while ((n = in.read(chunk)) != -1) buf.write(chunk, 0, n);
    byte[] data = buf.toByteArray();

    // Split header and body at \r\n\r\n
    int bodyStart = -1;
    for (int i = 0; i + 3 < data.length; i++) {
        if (data[i] == '\r' && data[i+1] == '\n' && data[i+2] == '\r' && data[i+3] == '\n') {
            bodyStart = i + 4;
            break;
        }
    }
    if (bodyStart >= 0) {
        try (FileOutputStream file = new FileOutputStream("homepage.html")) {
            file.write(data, bodyStart, data.length - bodyStart);
        }
    }
}
\end{lstlisting}

\textbf{3. Test Results \& Observations:}
\begin{itemize}
    \item \textbf{Terminal output:} Successfully connected to Google server IP \texttt{142.251.157.119:80}, received \texttt{HTTP/1.0 200 OK}, and saved \texttt{89,926 bytes} to \texttt{homepage.html}.
    \item \textbf{Browser view:} Opening \texttt{homepage.html} renders the offline Google homepage layout. Search actions and remote logo images rely on relative URLs (\texttt{/search}), confirming the static HTML nature of socket downloads.
\end{itemize}

\subsection*{3. Multithread in Java}
\noindent Create an application that has multiple threads running concurrently:
\begin{lstlisting}
class PrimeRun implements Runnable {
    long minPrime;
    PrimeRun ( long minPrime ) {
        this.minPrime = minPrime;
    }
    public void run() {
        // compute primes larger than minPrime
        . . .
    }

new Thread(p).start();

\begin{solutionbox}[title=Solution for Multithread Execution (PrimeRun.java)]
\begin{itemize}
    \item \textbf{Implementation:} Class \texttt{PrimeRun} computes primes exceeding \texttt{minPrime} in a loop guarded by \texttt{!Thread.currentThread().isInterrupted()}.
    \item \textbf{Concurrent Test:} In \texttt{main()}, two threads \texttt{T1} (min: 143) and \texttt{T2} (min: 1000) run simultaneously. Console outputs show interleaved prime values from both threads until interrupted after 20ms, demonstrating successful concurrency.
\end{itemize}
\end{solutionbox}

\subsubsection*{Stop a thread:}
\noindent\textit{Try to use \texttt{Thread.interrupt()} as in the following code and see what happens.}
\begin{lstlisting}
public void stop() {
    running = false;
    this.interrupt();
}
public void run() {
    running = true;
    while (running) {
        Socket s = socket.accept();
        ...
    }
}
\end{lstlisting}

\begin{solutionbox}[title=Answer \& Analysis: Why Thread.interrupt() Fails and How to Fix It]
\textbf{1. What happens when calling \texttt{stop()} with \texttt{this.interrupt()}?}
\begin{itemize}
    \item \textbf{Result:} \textbf{The thread DOES NOT stop! It remains blocked and alive indefinitely.}
\end{itemize}

\textbf{2. Root Cause Analysis:}
\begin{enumerate}
    \item \textbf{Blocking I/O Call:} \texttt{ServerSocket.accept()} is a synchronous blocking operation. When no incoming client connection arrives, the thread is suspended at \texttt{accept()} by the operating system kernel.
    \item \textbf{Interrupt Ignored by Blocking Sockets:} In standard Java I/O, \texttt{accept()} does not throw \texttt{InterruptedException}. Calling \texttt{Thread.interrupt()} only sets the thread's interrupt status flag, which \texttt{accept()} completely ignores.
    \item \textbf{Loop Never Re-evaluates:} Because execution is blocked inside \texttt{accept()}, the thread never reaches the top of the \texttt{while (running)} loop to evaluate \texttt{running == false}.
\end{enumerate}

\textbf{3. Correct Solution (\texttt{StopTest.java}):}
\begin{itemize}
    \item To unblock a thread stuck in \texttt{accept()}, we must close the underlying server socket:
    $$\texttt{serverSocket.close();}$$
    \item This forces \texttt{accept()} to immediately unblock and throw a \texttt{SocketException: Socket closed}. The catch block catches it, allowing the loop to exit cleanly.
\end{itemize}

\textbf{4. Experimental Verification Output:}
\begin{verbatim}
Testing running = false and t.interrupt()...
After interrupt, is thread alive? true
Testing serverSocket.close()...
accept() exited with: java.net.SocketException: Socket closed
Thread has stopped.
After close, is thread alive? false
\end{verbatim}

\subsubsection*{Multithread in Server Application:}
\begin{itemize}
    \item \textbf{Main window thread:} Runs Swing Event Dispatch Thread (GUI).
    \item \textbf{Child thread (Listener):} Runs \texttt{serverSocket.accept()} in a loop to accept connections asynchronously without freezing the GUI.
    \item \textbf{Grand-child thread (\texttt{ClientHandler}):} Spawned per connected client to manage data exchange and broadcast messages.
\end{itemize}

\subsubsection*{Multithread in Client Application:}
\begin{itemize}
    \item \textbf{Main window thread:} Manages the connection setup dialog.
    \item \textbf{Chat window thread:} Hosts the active chat window.
    \item \textbf{Grand-child thread (Reader):} Background thread continuously reading from the socket \texttt{InputStream} and updating the UI via \texttt{SwingUtilities.invokeLater()}.
\end{itemize}

\subsubsection*{Exercise 3:}
\noindent\textit{Using multithread programming model to make the chat application can talk to many different users concurrently.}

\begin{solutionbox}[title=Solution for Exercise 3: Concurrent Multi-User Chat Application]
\textbf{1. Architecture \& Concurrency Highlights:}
\begin{itemize}
    \item \textbf{Server Broadcast System:} Maintains an active client pool using \texttt{CopyOnWriteArrayList<ClientHandler>}. When a client sends a message, \texttt{broadcast()} relays it in real time to all connected clients.
    \item \textbf{UTF-8 Encoding:} Reads and writes use \texttt{UTF-8} character streams, fully preserving Vietnamese accented characters.
    \item \textbf{Clean Lifecycle Management:} When a chat window is closed, the client socket is gracefully closed, triggering server-side cleanup and notifying remaining users (\texttt{*** [user] left ***}).
\end{itemize}

\textbf{2. Live Testing Demonstration:}
\begin{itemize}
    \item \textbf{Setup:} \texttt{ChatServer} initialized on port 9999. Three simultaneous clients joined: \textbf{Thái}, \textbf{Sơn}, và \textbf{Bảo}.
    \item \textbf{Broadcast Verification:} Messages sent from any client (e.g., \textit{"Xin chào mọi người!"}, \textit{"Chào Thái nhé!"}) are instantly received and rendered across all 3 client windows and the server log in real time without thread locks or lag.
\end{itemize}
